## Python 统计（primary K 总数 ↔ 章目录出现次数）

在工程根 `wujun-os/` 执行：

```bash
python3 << 'PY'
import csv
from pathlib import Path

ROOT = Path(".")
ASSIGN = ROOT / "ledger/P4_归属表.csv"
PILLARS = ["D06", "D01", "D02"]

for p in PILLARS:
    primary = [
        r["card_id"]
        for r in csv.DictReader(ASSIGN.open(encoding="utf-8"))
        if r["primary_pillar"].startswith(f"{p}-")
    ]
    mds = sorted((ROOT / f"drafts/p5/{p}").glob("*.md"))
    blob = "\n".join(f.read_text(encoding="utf-8") for f in mds)
    mapped = sum(1 for c in primary if c in blob)
    occ = sum(blob.count(c) for c in primary)
    print(f"{p}: primary_K={len(primary)}, mapped={mapped}, gap={len(primary)-mapped}, occurrences={occ}, md_files={len(mds)}")
PY
```
